구글 연동 설치 안내 (Apps Script)

  1. 사용할 구글 스프레드시트를 생성하거나 엽니다.
  2. 상단 메뉴에서 [확장 프로그램] ➔ [Apps Script]를 클릭합니다.
  3. 기존 코드(Code.gs)를 지우고 아래 아래 소스코드를 전체 복사하여 붙여넣습니다.
  4. 우측 상단 [배포] ➔ [새 배포]를 클릭합니다.
  5. 톱니바퀴 ➔ 웹 앱 선택:
    • 실행 주체: 나 (Me)
    • 액세스 권한: 모든 사용자 (Anyone)
  6. 배포 후 생성된 웹 앱 URL을 본 프로그램의 설정창에 입력해주세요!
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // 헤더가 없으면 자동 생성
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        '작성일시', 
        '학년/반', 
        '번호', 
        '이름', 
        '관찰메모 및 인상깊은 장면', 
        '학습 및 수업태도 영역', 
        '정서 및 인성 영역', 
        '최종 행동특성 및 종합의견'
      ]);
      sheet.getRange(1, 1, 1, 8).setFontWeight('bold').setBackground('#e2e8f0');
    }
    
    sheet.appendRow([
      data.date || new Date().toLocaleString('ko-KR'),
      data.gradeClass || '',
      data.studentNum || '',
      data.studentName || '',
      data.observationNote || '',
      data.academicOutput || '',
      data.emotionalOutput || '',
      data.finalOutput || ''
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ result: 'success' }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ result: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
